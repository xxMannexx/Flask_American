@app.route("/video_recibido", methods=["POST", "GET"])
def video_recibido():
    if 'imagen' not in request.files:
        return jsonify({'error': 'No se encontró el archivo'}), 400

    imagen = request.files['imagen']
    usuario_id = 'Osw123'

    carpeta_usuario = os.path.join('imagenes_usuarios', usuario_id)
    os.makedirs(carpeta_usuario, exist_ok=True)

    nombre_archivo = usuario_id + datetime.datetime.now().strftime("%Y%m%d%H%M%S") + ".jpeg"
    ruta_completa = os.path.join(carpeta_usuario, nombre_archivo)

    imagen.save(ruta_completa)
    return jsonify({'mensaje': f'Imagen guardada en {ruta_completa}'})

@app.route("/transferencia_ruta", methods=["POST", "GET"])
def transferencia_ruta():
    if request.method == 'POST':
        destinatario = request.form['Tarjeta_Destinatario']

        if destinatario:
            cursor = db.database.cursor()
            consulta = 'INSERT INTO f (destino) VALUES (%s)'
            data = (destinatario,)
            cursor.execute(consulta, data)
            db.database.commit()

    return render_template('Transferencias.html')
